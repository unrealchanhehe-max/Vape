local espEnabled = false
local highlights = {}

local function removeESP(player)
    if highlights[player] then
        highlights[player]:Destroy()
        highlights[player] = nil
    end
end

local function addESP(player)
    if player == localPlayer then return end
    removeESP(player)

    local character = player.Character
    if not character then return end

    local highlight = Instance.new("Highlight")
    highlight.Name = "ESP"
    highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    highlight.FillTransparency = 0.5
    highlight.OutlineTransparency = 0
    highlight.Adornee = character
    highlight.Parent = character
    highlights[player] = highlight
end

local function setupPlayer(player)
    player.CharacterAdded:Connect(function()
        if espEnabled then
            task.wait(0.1)
            addESP(player)
        end
    end)
end

for _, p in ipairs(Players:GetPlayers()) do
    setupPlayer(p)
end
Players.PlayerAdded:Connect(setupPlayer)
Players.PlayerRemoving:Connect(removeESP)

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local camera = workspace.CurrentCamera
local localPlayer = Players.LocalPlayer

local aimEnabled = false
local smoothness = 0.2     -- 0.01 = rất mượt/chậm, 1 = khóa ngay
local maxRadius = 250      -- chỉ aim người trong vòng tròn này (pixel, tính từ tâm màn hình)
local targetPart = "Head"

local function getClosestTarget()
    local closest, shortest = nil, maxRadius
    local center = Vector2.new(camera.ViewportSize.X / 2, camera.ViewportSize.Y / 2)

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= localPlayer and player.Character then
            local humanoid = player.Character:FindFirstChildOfClass("Humanoid")
            local part = player.Character:FindFirstChild(targetPart)

            if humanoid and humanoid.Health > 0 and part then
                local screenPos, onScreen = camera:WorldToViewportPoint(part.Position)
                if onScreen then
                    local dist = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                    if dist < shortest then
                        shortest = dist
                        closest = part
                    end
                end
            end
        end
    end

    return closest
end

RunService.RenderStepped:Connect(function()
    if not aimEnabled then return end

    local target = getClosestTarget()
    if target then
        local goal = CFrame.lookAt(camera.CFrame.Position, target.Position)
        camera.CFrame = camera.CFrame:Lerp(goal, smoothness)
    end
end)

            end

            controls = require(player.PlayerScripts:WaitForChild("PlayerModule")):GetControls()
        end)
        if controls then controls:Disable() end

        local rx, ry = camera.CFrame:ToOrientation()
        pitch, yaw = rx, ry
        camera.CameraType = Enum.CameraType.Scriptable

        local conn
        conn = RunService.RenderStepped:Connect(function(dt)
            -- Xoay camera khi giữ chuột phải
            if UIS:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then
                UIS.MouseBehavior = Enum.MouseBehavior.LockCurrentPosition
                local delta = UIS:GetMouseDelta()
                yaw = yaw - delta.X * 0.003
                pitch = math.clamp(pitch - delta.Y * 0.003, -1.5, 1.5)
            else
                UIS.MouseBehavior = Enum.MouseBehavior.Default
            end

            local rot = CFrame.fromOrientation(pitch, yaw, 0)

            -- Di chuyển
            local move = Vector3.zero
            if UIS:IsKeyDown(Enum.KeyCode.W) then move += Vector3.new(0, 0, -1) end
            if UIS:IsKeyDown(Enum.KeyCode.S) then move += Vector3.new(0, 0, 1) end
            if UIS:IsKeyDown(Enum.KeyCode.A) then move += Vector3.new(-1, 0, 0) end
            if UIS:IsKeyDown(Enum.KeyCode.D) then move += Vector3.new(1, 0, 0) end
            if UIS:IsKeyDown(Enum.KeyCode.E) then move += Vector3.new(0, 1, 0) end
            if UIS:IsKeyDown(Enum.KeyCode.Q) then move += Vector3.new(0, -1, 0) end

            local currentSpeed = speed
            if UIS:IsKeyDown(Enum.KeyCode.LeftShift) then
                currentSpeed = speed * 3
            end

            local pos = camera.CFrame.Position + rot:VectorToWorldSpace(move) * currentSpeed * dt
            camera.CFrame = CFrame.new(pos) * rot
        end)

        -- Hàm tắt free cam
        local function stop()
            conn:Disconnect()
            UIS.MouseBehavior = Enum.MouseBehavior.Default
            camera.CameraType = Enum.CameraType.Custom
            local char = player.Character
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            if hum then camera.CameraSubject = hum end
            if controls then controls:Enable() end
        end

        if getgenv then getgenv().FreeCamStop = stop end
    end,
})

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local player = Players.LocalPlayer

local antiFling = false
local conn
local lastSafeCFrame

local MAX_SPEED = 150       -- vận tốc ngang tối đa được coi là bình thường
local MAX_SPIN = 40         -- tốc độ xoay tối đa

local function start()
    conn = RunService.Stepped:Connect(function()
        -- 1) Tắt va chạm với người chơi khác (kẻ fling chỉ có tác dụng khi va chạm được)
        for _, other in ipairs(Players:GetPlayers()) do
            if other ~= player and other.Character then
                for _, part in ipairs(other.Character:GetDescendants()) do
                    if part:IsA("BasePart") and part.CanCollide then
                        part.CanCollide = false
                    end
                end
            end
        end

        -- 2) Chặn vận tốc bất thường trên nhân vật của mình
        local char = player.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if not (root and hum) or hum.Health <= 0 then return end

        local vel = root.AssemblyLinearVelocity
        local horizontal = Vector3.new(vel.X, 0, vel.Z).Magnitude
        local limit = math.max(MAX_SPEED, hum.WalkSpeed * 3)

        if horizontal > limit or root.AssemblyAngularVelocity.Magnitude > MAX_SPIN then
            root.AssemblyLinearVelocity = Vector3.zero
            root.AssemblyAngularVelocity = Vector3.zero
            if lastSafeCFrame then
                root.CFrame = lastSafeCFrame
            end
        else
            lastSafeCFrame = root.CFrame
        end
    end)
end

local function stop()
    if conn then
        conn:Disconnect()
        conn = nil
    end
end

tab:CreateToggle({
    name = "Anti Fling",
    value = false,
    callback = function(state)
        antiFling = state
        if state then start() else stop() end
    end,
})

local Players = game:GetService("Players")

local hitboxEnabled = false
local hitboxSize = 10
local hitboxTransparency = 0.7
local originals = {}   -- lưu kích thước gốc để khôi phục

local function resetPart(root)
    local data = originals[root]
    if data and root.Parent then
        root.Size = data.Size
        root.Transparency = data.Transparency
        root.CanCollide = data.CanCollide
    end
    originals[root] = nil
end

local function resetAll()
    for root in pairs(originals) do
        resetPart(root)
    end
end

RunService.RenderStepped:Connect(function()
    if not hitboxEnabled then return end

    for _, other in ipairs(Players:GetPlayers()) do
        if other ~= player and other.Character then
            local root = other.Character:FindFirstChild("HumanoidRootPart")
            local hum = other.Character:FindFirstChildOfClass("Humanoid")

            if root and hum and hum.Health > 0 then
                if not originals[root] then
                    originals[root] = {
                        Size = root.Size,
                        Transparency = root.Transparency,
                        CanCollide = root.CanCollide,
                    }
                end
                root.Size = Vector3.new(hitboxSize, hitboxSize, hitboxSize)
                root.Transparency = hitboxTransparency
                root.CanCollide = false
                root.Massless = true
            end
        end
    end
end)

-- Dọn dữ liệu khi người chơi rời game
Players.PlayerRemoving:Connect(function(p)
    if p.Character then
        local root = p.Character:FindFirstChild("HumanoidRootPart")
        if root then originals[root] = nil end
    end
end)

-- Ẩn/hiện toàn bộ nhân vật (phía bạn, ai cũng sẽ thấy tuỳ cách game đồng bộ)
local function setInvisible(char, invisible)
    for _, obj in ipairs(char:GetDescendants()) do
        if obj:IsA("BasePart") and obj.Name ~= "HumanoidRootPart" then
            obj.LocalTransparencyModifier = invisible and 1 or 0
        elseif obj:IsA("Decal") or obj:IsA("Texture") then
            obj.Transparency = invisible and 1 or 0
        end
    end
end

local function start()
    if conn then conn:Disconnect() end
    conn = RunService.RenderStepped:Connect(function()
        local char = player.Character
        if char then
            setInvisible(char, true)
        end
    end)
end

local function stop()
    if conn then
        conn:Disconnect()
        conn = nil
    end
    local char = player.Character
    if char then
        setInvisible(char, false)
    end
end
